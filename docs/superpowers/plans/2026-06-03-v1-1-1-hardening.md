```go
// walCheckpointModes is the complete set of accepted SQLite WAL checkpoint modes.
var walCheckpointModes = map[string]struct{}{
	"PASSIVE":  {},
	"FULL":     {},
	"RESTART":  {},
	"TRUNCATE": {},
}

func (s *DB) WALCheckpoint(ctx context.Context, mode string) error {
	const op = "storage.sqlite.DB.WALCheckpoint"
	if mode == "" {
		mode = "TRUNCATE"
	}
	if _, ok := walCheckpointModes[mode]; !ok {
		return apperr.Newf(op, "invalid checkpoint mode: %q", mode)
	}
	_, err := s.conn.ExecContext(ctx, "PRAGMA wal_checkpoint("+mode+");")
	return apperr.Wrap(op, err)
}

func (s *DB) ExportHotSnapshot(ctx context.Context, targetPath string) error {
	const op = "storage.sqlite.DB.ExportHotSnapshot"
	if targetPath == "" {
		return apperr.New(op, "targetPath must not be empty")
	}
	if !filepath.IsAbs(targetPath) {
		return apperr.Newf(op, "targetPath must be absolute, got %q", targetPath)
	}
	clean := filepath.Clean(targetPath)
	if clean != targetPath {
		return apperr.Newf(op, "targetPath contains traversal, got %q", targetPath)
	}
	if err := os.MkdirAll(filepath.Dir(clean), 0755); err != nil {
		return apperr.Wrap(op, err)
	}
	if _, err := s.conn.ExecContext(ctx, "VACUUM INTO '"+clean+"'"); err != nil {
		return apperr.Wrap(op, err)
	}
	return nil
}
```

```go
func (s *DB) RotateBackups(backupDir string, keep int) error {
	const op = "storage.sqlite.DB.RotateBackups"
	if keep <= 0 {
		return nil
	}
	entries, err := os.ReadDir(backupDir)
	if err != nil {
		if os.IsNotExist(err) {
			return nil
		}
		return apperr.Wrap(op, err)
	}
	sort.Slice(entries, func(i, j int) bool {
		ii, _ := entries[i].Info()
		jj, _ := entries[j].Info()
		return ii.ModTime().After(jj.ModTime())
	})
	var errs []error
	for _, entry := range entries[keep:] {
		if err := os.Remove(filepath.Join(backupDir, entry.Name())); err != nil {
			errs = append(errs, err)
		}
	}
	if len(errs) > 0 {
		return apperr.Newf(op, "failed to remove %d backup(s): %v", len(errs), errs[0])
	}
	return nil
}
```

## Task 10: Phase 4 — Admin Token Verdict and Fix

**Files:**
- Modify: `cmd/cf-sync/daemon_runtime.go`
- Create: `docs/audits/ADMIN_TOKEN_FINAL_VERDICT.md`

### Context — CONFIRMED VULNERABILITY

`newAuthenticator()` in `daemon_runtime.go:48` hardcodes the token string `"admin-token"` in the token map. The `startAPIServer()` binds to `metricsAddr` which defaults to `":9090"` — all network interfaces. The `middleware.Auth` middleware validates the Bearer token against this hardcoded map.

Anyone on the network who can reach port 9090 can authenticate as admin using `Authorization: Bearer admin-token` and gain `runtime.execute`, `runtime.rollback`, `quarantine.manage`, and `audit.read` scopes.

**Two-part fix:**
1. Change the default bind to `127.0.0.1:9090` (loopback-only).
2. Load the API token from env var `CF_SYNC_API_TOKEN`; fail startup in daemon mode if not set.
