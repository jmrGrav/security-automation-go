# V1.6 ENV Elimination Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make `runtime.db` (SQLite) the single source of truth for all runtime feature flags, eliminating operational dependence on `/etc/security-automation-go/security-automation.env`.

**Architecture:** A `RuntimeFlags` struct read per-cycle from `ui_settings` in SQLite drives feature enable/disable without restart. The admin API server failure is demoted from a fatal daemon exit to a non-fatal warning. A new `/settings/runtime` UI page exposes all flags for operator control.

**Tech Stack:** Go 1.22, `modernc.org/sqlite`, `net/http` with `text/template` (no external frontend framework), Templ for typed templates.

## Critical architectural decisions — settled

**Decision A — Dynamic reload mechanism:** Runtime flags are re-read from SQLite at the start of each scheduler cycle via a `FlagsReader` passed into the orchestrator tick. Workers are always started but no-op when their flag is off. This avoids goroutine lifecycle management.

**Decision B — Secrets are NOT hot-reloadable in v1.6.** API tokens (Cloudflare, CrowdSec, AbuseIPDB, BetterStack) require a process restart to pick up new values. Document this as a known limitation. The plan does not change secret loading behavior.

**Decision C — Legacy path `/etc/security-automation/` (no `-go`):** Remove the legacy default `"/etc/security-automation/secrets"` from `health/checks.go:432` and `health_page.go:59`. Keep `startuplog/layout_check.go`'s `DefaultLegacyRoot` and warning logic (it is detection/warning code, not a loading path — keeping it is conservative and safe).

**Decision D — Env flag removal scope:** Remove from `applyEnvOverrides` in `config.go`: `CLOUDFLARE_MUTATIONS_ENABLED`, `CS_POLLER_ENABLED`, `ABUSEIPDB_ENABLED`, `ABUSEIPDB_REPORTING_ENABLED`. Keep: `UI_ADDR`, `STATE_DIR`, `METRICS_ADDR`, all secret env vars (tokens/keys), `UI_ENABLED`. The `DefaultEnvFile` constant and `LoadEnvFile` stay for the remaining non-flag vars.

func newDaemonContext(ctx context.Context, logger *slog.Logger, srv *http.Server) (context.Context, context.CancelFunc) {
    childCtx, cancel := context.WithCancel(ctx)
    sigChan := make(chan os.Signal, 1)
    signal.Notify(sigChan, syscall.SIGINT, syscall.SIGTERM)

    go func() {
        sig := <-sigChan
        logger.Info("received signal, shutting down", "signal", sig)

        if srv != nil {
            shutdownCtx, shutCancel := context.WithTimeout(context.Background(), 5*time.Second)
            defer shutCancel()
            _ = srv.Shutdown(shutdownCtx)
        }
        cancel()
    }()

    return childCtx, cancel
}

```go
// RuntimeFlags holds feature-enable flags stored in ui_settings.
// These are the only operational flags that survive env elimination.
type RuntimeFlags struct {
    CSPollerEnabled            bool
    CloudflareMutationsEnabled bool
    AbuseIPDBEnabled           bool
    BetterStackEnabled         bool
}

// GetRuntimeFlags reads all four feature flags from ui_settings.
// Missing rows default to false (safe: off by default, operator must enable).
func (s *SetupStore) GetRuntimeFlags(ctx context.Context) (RuntimeFlags, error) {
    keys := []string{
        "cs_poller_enabled",
        "cloudflare_mutations_enabled",
        "abuseipdb_enabled",
        "betterstack_enabled",
    }
    var flags RuntimeFlags
    for _, key := range keys {
        v, ok, err := s.GetSetting(ctx, key)
        if err != nil {
            return flags, err
        }
        if !ok {
            continue
        }
        enabled := v == "true"
        switch key {
        case "cs_poller_enabled":
            flags.CSPollerEnabled = enabled
        case "cloudflare_mutations_enabled":
            flags.CloudflareMutationsEnabled = enabled
        case "abuseipdb_enabled":
            flags.AbuseIPDBEnabled = enabled
        case "betterstack_enabled":
            flags.BetterStackEnabled = enabled
        }
    }
    return flags, nil
}

// SetRuntimeFlag persists a single feature flag.
// Valid keys: cs_poller_enabled, cloudflare_mutations_enabled, abuseipdb_enabled, betterstack_enabled.
func (s *SetupStore) SetRuntimeFlag(ctx context.Context, key string, enabled bool) error {
    const op = "storage.sqlite.SetupStore.SetRuntimeFlag"
    allowed := map[string]bool{
        "cs_poller_enabled":             true,
        "cloudflare_mutations_enabled":  true,
        "abuseipdb_enabled":             true,
        "betterstack_enabled":           true,
    }
    if !allowed[key] {
        return fmt.Errorf("%s: unknown flag key %q", op, key)
    }
    v := "false"
    if enabled {
        v = "true"
    }
    return s.SetSetting(ctx, key, v)
}
```

func (s *Server) handleRuntimeSettings(w http.ResponseWriter, r *http.Request) {
    if s.setupStore == nil {
        http.Error(w, "setup store not available", http.StatusServiceUnavailable)
        return
    }

    flags, err := s.setupStore.GetRuntimeFlags(r.Context())
    if err != nil {
        if s.logger != nil {
            s.logger.Error("get runtime flags", "err", err)
        }
        http.Error(w, "internal error", http.StatusInternalServerError)
        return
    }

    csrfToken := s.getOrSetCSRFToken(w, r)
    w.Header().Set("Content-Type", "text/html; charset=utf-8")
    fmt.Fprintf(w, runtimeSettingsPage(csrfToken, flags, ""))
}

func (s *Server) handleRuntimeSettingsPost(w http.ResponseWriter, r *http.Request) {
    if !s.validCSRF(r) {
        http.Error(w, "csrf required", http.StatusForbidden)
        return
    }
    if s.setupStore == nil {
        http.Error(w, "setup store not available", http.StatusServiceUnavailable)
        return
    }
    if err := r.ParseForm(); err != nil {
        http.Error(w, "bad request", http.StatusBadRequest)
        return
    }

    flags := map[string]bool{
        "cs_poller_enabled":            r.FormValue("cs_poller_enabled") == "true",
        "cloudflare_mutations_enabled": r.FormValue("cloudflare_mutations_enabled") == "true",
        "abuseipdb_enabled":            r.FormValue("abuseipdb_enabled") == "true",
        "betterstack_enabled":          r.FormValue("betterstack_enabled") == "true",
    }
    for key, enabled := range flags {
        if err := s.setupStore.SetRuntimeFlag(r.Context(), key, enabled); err != nil {
            if s.logger != nil {
                s.logger.Error("set runtime flag", "key", key, "err", err)
            }
            http.Error(w, "internal error", http.StatusInternalServerError)
            return
        }
    }

    if s.audit != nil {
        s.audit.Record("runtime_flags_updated", map[string]string{

package ui

import (
    "context"

    "net/http"
    "time"

type componentStatus struct {
    Name        string
    Status      string // "enabled", "disabled", "unconfigured"
    Level       string // "ok", "warn", "error"
    LastRun     string
    LastError   string
    LastSuccess string
}

func (s *Server) handleRuntimeStatus(w http.ResponseWriter, r *http.Request) {
    ctx := r.Context()

    var flags sqlite.RuntimeFlags
    if s.setupStore != nil {
        flags, _ = s.setupStore.GetRuntimeFlags(ctx)
    }

    credConfigured := func(key string) bool {
        if s.credentialStore == nil {
            return false
        }
        return credentialConfigured(ctx, s.credentialStore, key)
    }

    components := []componentStatus{
        {
            Name:   "CrowdSec Poller",
            Status: componentEnabledStatus(flags.CSPollerEnabled, credConfigured("crowdsec.lapi_key")),
            Level:  componentLevel(flags.CSPollerEnabled, credConfigured("crowdsec.lapi_key")),
        },
        {
            Name:   "Cloudflare Mutations",
            Status: componentEnabledStatus(flags.CloudflareMutationsEnabled, credConfigured("cloudflare.api_token")),
            Level:  componentLevel(flags.CloudflareMutationsEnabled, credConfigured("cloudflare.api_token")),
        },
        {
            Name:   "AbuseIPDB Reporter",
            Status: componentEnabledStatus(flags.AbuseIPDBEnabled, credConfigured("abuseipdb.api_key")),
            Level:  componentLevel(flags.AbuseIPDBEnabled, credConfigured("abuseipdb.api_key")),
        },
        {
            Name:   "BetterStack Telemetry",
            Status: componentEnabledStatus(flags.BetterStackEnabled, credConfigured("betterstack.source_token")),
            Level:  componentLevel(flags.BetterStackEnabled, credConfigured("betterstack.source_token")),
        },
    }

    csrfToken := s.getOrSetCSRFToken(w, r)
    w.Header().Set("Content-Type", "text/html; charset=utf-8")
    fmt.Fprintf(w, runtimeStatusPage(csrfToken, components, time.Now().UTC().Format(time.RFC3339)))
}

func componentEnabledStatus(enabled, credOK bool) string {
    if !credOK {
        return "unconfigured"
    }
    if enabled {
        return "enabled"
    }
    return "disabled"
}

func componentLevel(enabled, credOK bool) string {
    if !credOK {
        return "warn"
    }
    if enabled {
        return "ok"
    }
    return "warn"
}

<header><strong>Operator UI</strong>
<div class="nav">
  <a href="/">Dashboard</a>
  <a href="/providers">Providers</a>
